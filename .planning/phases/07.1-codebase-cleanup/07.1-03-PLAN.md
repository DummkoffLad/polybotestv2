```python
"""Mixins providing common strategy behaviors.

These mixins are inherited by strategy classes to share common
functionality without deep inheritance hierarchies.

Usage:
    class MyStrategy(SkipHelperMixin, HourlyBudgetMixin, Strategy):
        def __init__(self):
            # Mixins initialize their own state
            SkipHelperMixin.__init__(self)
            HourlyBudgetMixin.__init__(self)
            # ... rest of init
"""
from decimal import Decimal
from datetime import datetime
from typing import Dict, Optional
from .base import TradeDecision


class SkipHelperMixin:
    """Provides skip() helper for creating skip decisions with tracking.

    Requires the inheriting class to have:
    - self.skips: int (initialized to 0)
    - self.skip_reasons: Dict[str, int] (initialized to {})
    """

    def __init__(self):
        """Initialize skip tracking. Call from strategy __init__."""
        self.skips = 0
        self.skip_reasons: Dict[str, int] = {}

    def _skip(self, reason: str) -> TradeDecision:
        """Create a SKIP decision and track the reason.

        Args:
            reason: Short reason code (e.g., 'no_price', 'market_cap')

        Returns:
            TradeDecision with action=SKIP and skip_reason set
        """
        self.skips += 1
        self.skip_reasons[reason] = self.skip_reasons.get(reason, 0) + 1
        return TradeDecision.skip(reason)


class HourlyBudgetMixin:
    """Provides hourly budget tracking and reset logic.

    Resets the hourly_budget_used counter when the hour changes.
    Used by strategies that limit deployment per hour.

    Requires the inheriting class to have:
    - self.hourly_budget_used: Decimal (initialized to 0)
    - self._current_hour: Optional[int] (initialized to None or current hour)
    """

    def __init__(self):
        """Initialize hourly tracking. Call from strategy __init__."""
        self.hourly_budget_used = Decimal("0")
        self._current_hour: Optional[int] = None

    def _check_hourly_reset(self, event_time: datetime) -> None:
        """Reset hourly budget if hour changed since last check.

        Args:
            event_time: Timestamp of current event
        """
        current_hour = event_time.hour
        if self._current_hour is not None and current_hour != self._current_hour:
            self.hourly_budget_used = Decimal("0")
        self._current_hour = current_hour
```

Important design notes:
- Each mixin has its own `__init__` that initializes only its state
- Strategies call each mixin's `__init__` explicitly in their `__init__`
- Private method names (`_skip`, `_check_hourly_reset`) preserved for backward compatibility
- Mixins are designed to be composed, not deeply inherited
