```python
"""Unified PnL calculation logic for strategies.

This module provides a single implementation of PnL calculation
used across all strategy implementations to ensure consistency.
"""
from decimal import Decimal
from typing import Dict, Tuple

from ..core.portfolio import Portfolio
from ..data.models import PriceSnapshot


def calculate_strategy_pnl(
    portfolio: Portfolio,
    final_prices: Dict[str, PriceSnapshot]
) -> Tuple[Decimal, Decimal]:
    """Calculate realized and unrealized PnL for a portfolio.

    Args:
        portfolio: Strategy's portfolio containing positions and realized PnL
        final_prices: Dictionary mapping token_id to final PriceSnapshot

    Returns:
        Tuple of (realized_pnl, unrealized_pnl) as Decimals

    Note:
        Unrealized PnL = sum of (shares * bid - cost_basis) for each position
        Realized PnL = portfolio.realized_pnl (already tracked by portfolio)

    Example:
        realized, unrealized = calculate_strategy_pnl(self.portfolio, final_prices)
        total_pnl = realized + unrealized
    """
    # Calculate unrealized PnL for all open positions
    unrealized = Decimal("0")
    for token_id, position in portfolio.get_positions().items():
        if position.shares <= 0:
            continue
        if token_id not in final_prices:
            continue
        price_snapshot = final_prices[token_id]
        if not price_snapshot.bid or price_snapshot.bid <= 0:
            continue
        # Unrealized = current value - cost basis
        unrealized += position.shares * price_snapshot.bid - position.cost_basis

    return portfolio.realized_pnl, unrealized
```

Key design decisions:
- Function takes portfolio and prices as arguments (no strategy dependency)
- Returns tuple for consistency with existing strategy interface
- Handles edge cases (no position, no price, zero shares)
- Documented with docstring and example
